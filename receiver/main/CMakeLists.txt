set(VIBE_RECEIVER_BOARD "c480" CACHE STRING "Receiver board: c480 or geek")
if(VIBE_RECEIVER_BOARD STREQUAL "geek")
  set(receiver_source "geek.cpp")
elseif(VIBE_RECEIVER_BOARD STREQUAL "c480")
  set(receiver_source "receiver.cpp")
else()
  message(FATAL_ERROR "Unknown VIBE_RECEIVER_BOARD: ${VIBE_RECEIVER_BOARD}")
endif()
idf_component_register(SRCS "${receiver_source}" "../../main/usb.c" "../../main/wireless.cpp" INCLUDE_DIRS "." "../../main" "../../private" REQUIRES M5GFX tinyusb nvs_flash esp_timer driver esp_lcd esp_psram usb esp_wifi esp_netif lwip)
target_compile_definitions(${COMPONENT_LIB} PRIVATE VIBE_RECEIVER=1)
if(VIBE_RECEIVER_BOARD STREQUAL "geek")
  target_compile_definitions(${COMPONENT_LIB} PRIVATE VIBE_RECEIVER_GEEK=1)
endif()
idf_component_get_property(tusb_lib tinyusb COMPONENT_LIB)
target_include_directories(${tusb_lib} PUBLIC "${CMAKE_CURRENT_LIST_DIR}/../../main")
if(DEFINED VIBE_DISPLAY_IDLE_MS)
  target_compile_definitions(${COMPONENT_LIB} PRIVATE VIBE_DISPLAY_IDLE_MS=${VIBE_DISPLAY_IDLE_MS})
endif()
